CREATE TABLE IF NOT EXISTS plants (
    plant_id INTEGER PRIMARY KEY,
    plant_name_scientific_reference TEXT,
    plant_name_scientific_reference_normalize TEXT
)

CREATE TABLE IF NOT EXISTS activities (
    activity_id INTEGER PRIMARY KEY,
    activity_name_reference TEXT,
    activity_name_reference_normalize TEXT
)

CREATE TABLE IF NOT EXISTS plants_activities (
    plant_activity_id INTEGER PRIMARY KEY,
    plant_id TEXT,
    activity_id TEXT
)

CREATE TABLE IF NOT EXISTS conditions (
    condition_id TEXT,
    condition_name_reference TEXT,
    condition_name_reference_normalize TEXT
)

CREATE TABLE IF NOT EXISTS plant_parts (
    plant_part_id TEXT,
    plant_part_name_reference TEXT,
    plant_part_name_reference_normalize TEXT,
